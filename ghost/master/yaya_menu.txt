OnMenu
{
	_kind='普通の時計'
	if clock_status=='pomodoro' {
		_kind=OnPomodoroStatus
	}
	elseif clock_status=='kitchen' {
		_kind='キッチンタイマー'
	}
	OnRefreshClock_Exec+'\![set,autoscroll,disable]\_q'
	--
	"現在:%(_kind)\n[200]"
	--
	if clock_status=='pomodoro' || clock_status=='kitchen' {
		TopMenu_AboutTimer
		--
		MenuItem_HR+右上の閉じるボタン+'\_q\e'
		return
	}
	else {
		TopMenuItem_Kitchen+'\n[200]'
		--
		TopMenuItem_Pomodoro+'\n[200]'
		--
		TopMenuItem_Alarm+'\n[200]'
		--
		TopMenuItem_Signal
	}
	--
	右上の閉じるボタン+'\_q\e'
}

MenuItem_HR
{
	_full_length=18
	_length=_full_length-GETSTRBYTES(_argv[0])/2
	for _hr='', _c=0; _c<=_length; _c++; {
		_hr+='─'
	}
	if STRLEN(_argv[0])<=0 {
		'\n[60]\_n\f[shadowcolor,none]\f[color,disable]'+_hr+'\f[default]\_n\n[-60]'
	}
	else {
		_argv[0]+'\_n\f[shadowcolor,none]\f[color,disable]'+_hr+'\f[default]\_n'
	}
}

OnPomodoroKindsConvert
{
	//定義を文字列で　数字系は後で数値型へ直す
	_n=('0','1','2');
	_e=('work','rest','longrest');
	_j=('作業','休憩','長めに休憩');

	//引数を番号へ変換
	_a0=TOSTR(_argv[0])
	if _a0==_n[1]||_a0==_e[1]||_a0==_j[1] { _a0=_n[1] }
	elseif _a0==_n[2]||_a0==_e[2]||_a0==_j[2] { _a0=_n[2] }
	else { _a0=_n[0] }
	_a0=TOINT(_a0)

	//指定された形式で返す
	if _argv[1]=='n'||_argv[1]=='num'||_argv[1]=='number' { TOINT(_n[_a0]) }
	elseif _argv[1]=='e'||_argv[1]=='eng'||_argv[1]=='english' { _e[_a0] }
	else { _j[_a0] }
}

OnAlarmStatusSet
{
	_r0=reference[0]
	_r2=reference[2]

	_t=STRFORM('$04d',TOSTR(ZEN2HAN(SUBSTR(_r0,0,4))))
	_h=TOINT(SUBSTR(_t,0,2))
	_m=TOINT(SUBSTR(_t,2,2))

	_c=alarm_status
	if alarm_active[0]==0 { _c[0]=0 }
	if alarm_active[1]==0 { _c[1]=0 }
	if alarm_active[2]==0 { _c[2]=0 }
	_c[_r2]=''

	if _h<0 || _h>23 || _m<0 || _m>59 {
		OnUkaclockRefresh_Exec
		--
		"\_q範囲外の数字か、数字以外が入力されました。\_q\n[150]\![embed,OnAlarmStatusDialog,%(_r2)]"
		return
	}
	elseif STRSTR(TOSTR(_c),_t,0)>=0 && alarm_active[_r2]==1 {
		OnUkaclockRefresh_Exec
		--
		"\_q同じ時刻を複数個有効にすることはできません。\_q\n[150]\![embed,OnAlarmStatusDialog,%(_r2)]"
		return
	}
	else
	{
		alarm_status[_r2]=_t
		OnMenu_AboutAlarm
	}
}

OnAlarmActiveChange
{
	_r0=reference[0]

	_c=alarm_status

	if alarm_active[2]==0{_c[2]=0}
	_c[_r0]=''

			OnUkaclockRefresh_Exec+"\_q同じ時刻を複数個有効にすることはできません。\_q\x"

		alarm_active[_r0]=0
	}
	--
	if ISFUNC(reference[1])==1 {
		"\![embed,%(reference[1])]\e"
		return
	}
	OnMenu_AboutAlarm
}

AboutSignal_Item
{
	_a0=TOINT(_argv[0])
	_active=signal_active[_a0]
	_soundfile=signal_sound[_a0]
	_status=signal_status[_a0]
	_method=signal_notifymethod[_a0]
	_memo=signal_memo[_a0]
	_memolimit=16

	MenuItem_HR("◆時報設定%(HAN2ZEN(_a0+1))")+'\n'
	--
	"\__q[OnSignalActiveChange,%(_a0)]状態:"
	--
	if _active==1 {'[有効]'} else {'[無効]'}
	--
	"\__q \__q[OnSignalStatusDialog,%(_a0)]毎時:[%(_status)]\__q "
	--
	_repeat='なし'
	if signal_repeat[_a0]==1 { _repeat='あり' }
	"\__q[OnSignalRepeatToggle,%(_a0)]繰り返し:[%(_repeat)]\__q\n"
	--
	"\__q[OnSignalNotifyMethodToggle,%(_a0)]通知方法:"
	--
	if _method==0 {
		"[ OS通知 ]\__q\__q[OnSignalNotifyTest,%(_a0)][試行]\__q"
	}
	elseif _method==1 {
		_memolimit=5
		"[ 音のみ ]\__q"
		--
		if FSIZE(_soundfile)>0 {
			_soundfile=SPLITPATH(_soundfile)
			_soundfile=_soundfile[2]
			"\__q[OnSignalNotifyTest,%(_a0)][試聴]\__q"
		}
		else {
			_soundfile='未設定'
			"\f[color,disable]\f[shadowcolor,none][試聴]\f[default]"
		}
		--
		"\__q[OnSignalSoundDialog,%(_a0)]♪:[%(_soundfile)]\__q"
	}
	else {
		"[バルーン]\__q\__q[OnSignalNotifyTest,%(_a0)][試行]\__q"
	}
	--
	if STRLEN(_memo)<=0 {
		" \__q[OnSignalMemoDialog,%(_a0)]メモ:[\f[color,disable]\f[shadowcolor,none]なし\f[default]]\__q"
	}
	else {
		" \__q[OnSignalMemoDialog,%(_a0)]メモ:[%(SHIORI3FW.MakeShortText(signal_memo[_a0],_memolimit))]\__q"
	}
}

OnSignalStatusSet
{
	_r0=reference[0]
	_r2=reference[2]

	_t=STRFORM('$02d',ZEN2HAN(SUBSTR(_r0,0,2)))
	_m=TOINT(_t)

	_c=signal_status
	if signal_active[0]==0{_c[0]=0}
	if signal_active[1]==0{_c[1]=0}
	if signal_active[2]==0{_c[2]=0}
	_c[_r2]=''

	if ( _m<0 || _m>59 ) {
		OnUkaclockRefresh_Exec
		--
		"\_q範囲外の数字か、数字以外が入力されました。\_q\n[150]\![embed,OnSignalStatusDialog,%(_r2)]"
		return
	}
	elseif STRSTR(TOSTR(_c),TOSTR(_m),0)!=-1 && signal_active[_r2]==1 {
		OnUkaclockRefresh_Exec
		--
		"\_q同じ時刻を複数個有効にすることはできません。\_q\n[150]\![embed,OnSignalStatusDialog,%(_r2)]"
		return
	}
	else
	{
		signal_status[_r2]=STRFORM('$02d',_m)
		OnMenu_AboutSignal
	}
}

		return
	}
	OnMenu_AboutSignal
