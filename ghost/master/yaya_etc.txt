//------------------------------------------------------------------------------
//インストール完了メッセージ
//------------------------------------------------------------------------------
OnInstallComplete
{
	_ref0 = SPLIT(reference[0],' with',2)[0] //hogehoge with hemohemo

	OnRefreshClock_Exec + '\_q'
	--
	if _ref0 == "shell" {
		"シェル"
	}
	elseif _ref0 == "ghost" {
		"ゴースト"
	}
	elseif _ref0 == "balloon" {
		"バルーン"
	}
	elseif _ref0 == "plugin" {
		"プラグイン"
	}
	--
	"「%(SHIORI3FW.EscapeAllTags(reference[1]))」、インストール完了しました。\n"
	--
	if basewarenameex == 'SSP' {
		_id = ''
		if _ref0 == "shell" {
			_id = 'shell'
		}
		elseif _ref0 == "ghost" {
			_id = 'ghost'
		}
		elseif _ref0 == "balloon" {
			_id = 'balloon'
		}
		
		if _id == 'ghost' {
			"切り替え・呼び出ししますか？"
			--
			AYATEMPLATE.MenuItem("切り替える","OnAYATEMPLATE.InstallChange",ghost,200)
			--
			AYATEMPLATE.MenuItem("呼び出す","OnAYATEMPLATE.InstallCall",ghost)
			--
			AYATEMPLATE.MenuItem("何もしない","Menu_CANCEL")
		}
		elseif _id != '' {
			"切り替えますか？"
			--
			AYATEMPLATE.MenuItem("切り替える","OnAYATEMPLATE.InstallChange",_id,200)
			--
			AYATEMPLATE.MenuItem("切り替えない","Menu_CANCEL")
		}
	}
	--
	'\_q\e'
}

OnAYATEMPLATE.InstallChange
{
	"\![change,%(reference[0]),lastinstalled]"
}

OnAYATEMPLATE.InstallCall
{
	'\![call,ghost,lastinstalled]'
}

//------------------------------------------------------------------------------
//ネットワーク更新失敗
//------------------------------------------------------------------------------
OnUpdateFailure
{
	OnRefreshClock_Exec + '\_q'
	--
	"ネットワーク更新に失敗しました。\n"
	--
	if reference[0] == "timeout"
	{
		//---- タイムアウトした
		"理由は「時間切れ」です。"
	}
	elseif reference[0] == "md5 miss"
	{
		//---- MD5値が一致しなかった
		"理由は「MD5値の不一致」です。"
		--
		OnUpdateFailure.houkokushimasuka
	}
	else
	{
		//---- 上記以外の理由で失敗した
		"理由は「%(reference[0])」です。"
		--
		OnUpdateFailure.houkokushimasuka
	}
	--
	"\e"
}

OnUpdateFailure.houkokushimasuka
{
	"\n[150]お手数をおかけしますが、ゴースト作者までご連絡いただけるとありがたいです。\n[150]"
	--
	AYATEMPLATE.MenuItem("報告するためにブラウザを開く","OnUpdateFailure.houkokusuru")
	--
	AYATEMPLATE.MenuItem("報告しない","script:\e")
}

OnUpdateFailure.houkokusuru
{
	OnRefreshClock_Exec
	--
	"\![open,browser,https://lnx.flop.jp/form/postmail.html]"
	--
	'\e'
}

//------------------------------------------------------------------------------
//時刻取得成功、比較開始
//この関数で 2008/03/25 19:15:05 のように見やすい形に作り変えます。
//詳しくはSTRFORM関数の説明を見てね！
//わからなければこのままで。
//------------------------------------------------------------------------------

SNTPCompare_StrForm
{
	_txt = SPLIT(_argv[0],',')
	STRFORM("$04d/$02d/$02d $02d:$02d:$02d",TOINT(_txt[0]),TOINT(_txt[1]),TOINT(_txt[2]),TOINT(_txt[3]),TOINT(_txt[4]),TOINT(_txt[5]))
}

OnSNTPCompare
{
	"%(OnRefreshClock_Exec)\_q/
	修正前 %(SNTPCompare_StrForm(reference[2]))\n/
	修正後 %(SNTPCompare_StrForm(reference[1]))\n/
	\_q\n"
	--
	if reference[3] == 0
	{
		"誤差はありませんでした。"
	}
	else
	{
		//\6は補正実行タグ。消しちゃだめ
		"\6%(OnRefreshClock_Exec)\n%(reference[3])秒のずれを修正しました。"
		--
		OnRefreshClock_Exec
	}
	--
	'\e'
}

//******************************************************************************
//キーが押された
//******************************************************************************
/*
//------------------------------------------------------------------------------
//OnKeyPressイベント
//------------------------------------------------------------------------------
//キーボードの任意のキーが押されたときに発動するイベントです。
//ファンクションキーを指定するときは、"f1"等と指定します。
//押されたキーはreference0に格納されます。
//if文、もしくはcase~when文で分岐させると良いでしょう。
*/

OnKeyPress
{
	case reference0 {
		when "r" {
			'\![reload,shiori]\![embed,OnRefreshClock_Exec]\_q(RELOAD SHIORI)\_q\e'
		}
		when "m" {
			OnMouseDoubleClick
		}
		when 'd' {
			if ISFUNC('OnDebugMenu')==1 { OnDebugMenu }
		}
	}
}
