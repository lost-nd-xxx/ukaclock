//******************************************************************************
// YAYA　ゴーストテンプレート
// 起動/終了トーク辞書
//******************************************************************************

OnFirstBoot
{
	OnBoot
}

//******************************************************************************
//通常の起動 OnBoot
//******************************************************************************

OnBoot
{
	OnSHIORI3FW.SurfaceRestore
}

//******************************************************************************
//他のゴーストから変更
//******************************************************************************

OnGhostChanged
{
	OnSHIORI3FW.SurfaceRestore
}

//******************************************************************************
//シェル変更後
//******************************************************************************

OnShellChanged
{
	OnSHIORI3FW.SurfaceRestore
}

//******************************************************************************
//最小化からの復帰 OnWindowStateRestore
//******************************************************************************

OnWindowStateRestore
{
	//長時間非表示していると通知が溜まっているかもなのでリセットする
	//時報・アラームはリセットに含まれない
	SetDefaultValuesForReset
	OnSHIORI3FW.SurfaceRestore
}

OnFullScreenAppRestore
{
	OnWindowStateRestore
}

OnSysResume
{
	OnWindowStateRestore
}
